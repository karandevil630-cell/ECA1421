#!/bin/bash

echo "Enter first number:"
read a

echo "Enter second number:"
read b

echo "Enter third number:"
read c

if [ $a -ge $b ] && [ $a -ge $c ]
then
    biggest=$a
elif [ $b -ge $a ] && [ $b -ge $c ]
then
    biggest=$b
else
    biggest=$c
fi

echo "Biggest number is $biggest"

output:
Enter first number:
25
Enter second number:
45
Enter third number:
30

Biggest number is 45
